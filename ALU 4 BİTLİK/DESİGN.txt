module alu_4bit (
    input [3:0] a,          // 4-bitlik 1. giriş (A)
    input [3:0] b,          // 4-bitlik 2. giriş (B)
    input [1:0] select,     // 2-bitlik seçim sinyali (Select: 00:Toplama, 01:Çıkarma, 10:AND, 11:OR)
    output reg [3:0] cikis, // 4-bitlik sonuç çıkışı
    output reg carry_out    // Toplama/Çıkarma için elde (taşma) sinyali
);

    always @(*) begin
        carry_out = 0; // Varsayılan değer
        case (select)
            2'b00: begin // Toplama (Add)
                {carry_out, cikis} = a + b;
            end
            
            2'b01: begin // Çıkarma (Subtract)
                {carry_out, cikis} = a - b;
            end
            
            2'b10: begin // AND İşlemi
                cikis = a & b;
            end
            
            2'b11: begin // OR İşlemi
                cikis = a | b;
            end
            
            default: begin
                cikis = 4'b0000;
            end
        endcase
    end

endmodule
