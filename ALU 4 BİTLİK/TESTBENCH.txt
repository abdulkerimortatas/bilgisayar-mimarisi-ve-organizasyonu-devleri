`timescale 1ns / 1ps

module tb_alu_4bit;
    // Testbench sinyalleri
    reg [3:0] a;
    reg [3:0] b;
    reg [1:0] select;
    wire [3:0] cikis;
    wire carry_out;

    // ALU modülünü bağlama (Instantiation)
    alu_4bit uut (
        .a(a),
        .b(b),
        .select(select),
        .cikis(cikis),
        .carry_out(carry_out)
    );

    initial begin
        // Dalga formu (waveform) çıktısı için gerekli tanımlar
        $dumpfile("dump.vcd");
        $dumpvars(1, tb_alu_4bit);

        // Test senaryoları
        $display("Zaman | Select |  A  |  B  | Çıkış | Carry");
        $display("----------------------------------------");

        // 1. Test: Toplama (Select = 00) -> 5 + 3 = 8
        a = 4'd5; b = 4'd3; select = 2'b00; #10;
        $display("%0t   |   %b   | %b | %b |  %b  |   %b", $time, select, a, b, cikis, carry_out);

        // 2. Test: Çıkarma (Select = 01) -> 7 - 4 = 3
        a = 4'd7; b = 4'd4; select = 2'b01; #10;
        $display("%0t   |   %b   | %b | %b |  %b  |   %b", $time, select, a, b, cikis, carry_out);

        // 3. Test: AND İşlemi (Select = 10) -> 12 (1100) & 10 (1010) = 8 (1000)
        a = 4'b1100; b = 4'b1010; select = 2'b10; #10;
        $display("%0t   |   %b   | %b | %b |  %b  |   %b", $time, select, a, b, cikis, carry_out);

        // 4. Test: OR İşlemi (Select = 11) -> 5 (0101) | 3 (0011) = 7 (0111)
        a = 4'b0101; b = 4'b0011; select = 2'b11; #10;
        $display("%0t   |   %b   | %b | %b |  %b  |   %b", $time, select, a, b, cikis, carry_out);

        $finish;
    end

endmodule
