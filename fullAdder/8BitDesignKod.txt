// 1-Bit Full Adder Modülü (Aynı kalır)
module full_adder (
    input logic a,
    input logic b,
    input logic cin,
    output logic s,
    output logic cout
);
    assign s = a ^ b ^ cin;
    assign cout = (a & b) | (cin & (a ^ b));
endmodule

// 8-Bit Ripple Carry Adder Modülü
module full_adder_8bit (
    input logic [7:0] a,
    input logic [7:0] b,
    input logic cin,
    output logic [7:0] s,
    output logic cout
);
    // 8 bitlik toplama için ara elde (carry) hatları (C1'den C7'ye)
    logic [7:1] c;

    // 8 adet 1-bitlik toplayıcının zincirleme bağlanması
    full_adder fa0 (.a(a[0]), .b(b[0]), .cin(cin),  .s(s[0]), .cout(c[1]));
    full_adder fa1 (.a(a[1]), .b(b[1]), .cin(c[1]), .s(s[1]), .cout(c[2]));
    full_adder fa2 (.a(a[2]), .b(b[2]), .cin(c[2]), .s(s[2]), .cout(c[3]));
    full_adder fa3 (.a(a[3]), .b(b[3]), .cin(c[3]), .s(s[3]), .cout(c[4]));
    full_adder fa4 (.a(a[4]), .b(b[4]), .cin(c[4]), .s(s[4]), .cout(c[5]));
    full_adder fa5 (.a(a[5]), .b(b[5]), .cin(c[5]), .s(s[5]), .cout(c[6]));
    full_adder fa6 (.a(a[6]), .b(b[6]), .cin(c[6]), .s(s[6]), .cout(c[7]));
    full_adder fa7 (.a(a[7]), .b(b[7]), .cin(c[7]), .s(s[7]), .cout(cout));

endmodule
