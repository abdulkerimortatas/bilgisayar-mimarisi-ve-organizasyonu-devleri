module alu_4bit (
    input  [3:0] A,          // 4-bit Giriş A
    input  [3:0] B,          // 4-bit Giriş B
    input  [1:0] ALU_Sel,    // 2-bit Seçim Sinyali
    output reg [7:0] ALU_Out // 8-bit Çıkış (Çarpma için taşmayı önler)
);

    always @(*) begin
        case(ALU_Sel)
            2'b00:   ALU_Out = A + B;        // Toplama
            2'b01:   ALU_Out = A - B;        // Çıkarma
            2'b10:   ALU_Out = A * B;        // ÇARPMA (4-bit * 4-bit = 8-bit)
            2'b11:   begin
                        if (B != 4'b0)
                            ALU_Out = A / B; // Bölme (Sıfıra bölme kontrolü)
                        else
                            ALU_Out = 8'b0;
                     end
            default: ALU_Out = 8'b0;
        endcase
    end

endmodule
